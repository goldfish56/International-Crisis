#africa

is_macro_region_west_africa = {
	or = {
		any_province = {
			owner = this
			is_capital = yes
			OR = {
				region = sr:region_senegal
				region = sr:region_niger
				region = sr:region_congo
				region = sr:region_north_africa
			}
		}
	}
}
